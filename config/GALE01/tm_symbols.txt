# rollback saves, and not referenced by Slippi, m-ex or TM-CE.
tm_state 804D646C

# Persistent training-mode state (TmPersist, src/melee/tm/tm.h). A block of

# NOT saved by Slippi's rollback and not zeroed by the OS; tm_Hook_Boot
